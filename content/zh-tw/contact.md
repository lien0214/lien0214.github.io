<script>
(function () {
  var SCRIPT_URL = '{{ site.Params.contactScriptURL }}';
  var CLIENT_ID  = '{{ site.Params.googleClientID }}';

  var form       = document.getElementById('cf-form');
  var submitBtn  = document.getElementById('cf-submit');
  var btnLabel   = document.getElementById('cf-btn-label');
  var btnSpinner = document.getElementById('cf-btn-spinner');
  var success    = document.getElementById('cf-success');
  var errorBox   = document.getElementById('cf-error');
  var msgError   = document.getElementById('cf-msg-error');
  var tokenClient;

  function setLoading(on) {
    btnLabel.style.display   = on ? 'none' : 'inline';
    btnSpinner.style.display = on ? 'inline' : 'none';
    submitBtn.disabled = on;
  }

  function showSuccess() {
    form.style.display    = 'none';
    success.style.display = 'block';
  }

  function showError() {
    setLoading(false);
    errorBox.style.display = 'block';
  }

  function sendMessage(name, email, message) {
    var honeypot = document.getElementById('cf-website').value;
    if (honeypot) { showSuccess(); return; }

    fetch(SCRIPT_URL, {
      method: 'POST',
      mode: 'no-cors',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: name, email: email, message: message, website: '' })
    }).then(showSuccess).catch(showError);
  }

  function initGSI() {
    tokenClient = google.accounts.oauth2.initTokenClient({
      client_id: CLIENT_ID,
      scope: 'email profile',
      callback: function (resp) {
        if (resp.error) { showError(); return; }
        fetch('https://www.googleapis.com/oauth2/v3/userinfo', {
          headers: { 'Authorization': 'Bearer ' + resp.access_token }
        })
        .then(function (r) { return r.json(); })
        .then(function (user) {
          sendMessage(user.name || '', user.email || '', form._pendingMsg);
        })
        .catch(showError);
      }
    });
  }

  submitBtn.addEventListener('click', function () {
    var message = document.getElementById('cf-message').value.trim();
    msgError.style.display = message ? 'none' : 'inline';
    if (!message) return;

    if (!CLIENT_ID || !SCRIPT_URL) { showError(); return; }

    setLoading(true);
    form._pendingMsg = message;

    if (tokenClient) {
      tokenClient.requestAccessToken({ prompt: 'select_account' });
    } else {
      showError();
    }
  });

  if (CLIENT_ID) {
    var s = document.createElement('script');
    s.src = 'https://accounts.google.com/gsi/client';
    s.onload = initGSI;
    s.async = true;
    document.head.appendChild(s);
  }
})();
</script>
